# One pkg per line; lines starting with # are ignored.
# Runtime packages — installed in the chroot and shipped in the ISO.
# Distinct from buildpkgs.txt, which is purged before slim.
#
# Every entry should name the build step (or runtime symptom) that
# demanded it, so future readers can audit whether the demand still
# holds.
# libs-base configure: GSXML / NSXMLNode / NSPropertyListSerialization
# XML branch. libgnustep-base.so links against libxml2.so at runtime.
libxml2

# libs-base configure: NSLocale / NSString unicode handling.
# libgnustep-base.so links against libicui18n.so + libicuuc.so at runtime.
icu

# Runtime dep surfaced at boot: ld-elf.so.1 fails to load
# libgnustep-base.so.1.31 because of "Shared object libffi.so.8 not
# found". libffi gets pulled in as a transitive of cmake/ninja at
# build time but removed by `pkg autoremove` after the buildpkgs purge.
# Pin it explicitly here so it stays on the ISO.
libffi

# GPU kernel modules. Loaded by kmodloader at boot when matching
# hardware is present (Phase 1+ of the freebsd-kmodloader plan); for
# now just shipped on the ISO so users can `kldload i915kms`,
# `kldload amdgpu`, etc. by hand. Pulls drm-kmod-firmware-* for the
# blobs many modern GPUs need.
drm-latest-kmod

# NVIDIA DRM driver: nvidia.ko + nvidia-modeset.ko + nvidia-drm.ko.
# kmodloader loads this when it detects vendor=0x10de in pciconf -l;
# loader.conf sets hw.nvidiadrm.modeset=1 so DRM/KMS engages on load.
# Restricted-distribution NVIDIA blob — pkg requires LICENSES_ACCEPTED=
# NVIDIA at install time (set in build.sh's pkg invocation env).
nvidia-drm-latest-kmod

# WiFi firmware blobs. Each wifi-firmware-*-kmod packages firmware-only,
# loaded automatically when the corresponding driver kld attaches:
#   wifi-firmware-iwlwifi-kmod  -> if_iwlwifi   (Intel AC8xxx/9xxx, AX, BE)
#   wifi-firmware-rtw88-kmod    -> if_rtw88     (Realtek Wi-Fi 5 era)
#   wifi-firmware-rtw89-kmod    -> if_rtw89     (Realtek Wi-Fi 6 / 6E / 7)
# Drivers themselves are in base.txz; kmodloader matches via
# personality plists in /System/Library/Extensions and kldloads when
# matching hardware is present.
wifi-firmware-iwlwifi-kmod
wifi-firmware-rtw88-kmod
wifi-firmware-rtw89-kmod
